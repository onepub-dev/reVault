use std::{fs,collections::BTreeMap};
fn med(v:&mut [f64])->f64{v.sort_by(f64::total_cmp);let n=v.len();(v[(n-1)/2]+v[n/2])/2.0}
fn main(){for name in ["read-small","read-large"]{let csv=fs::read_to_string(format!("revault_lockbox_api/benches/results/issue310-shared-window-{name}.csv")).unwrap();let mut groups=BTreeMap::<String,BTreeMap<usize,[f64;2]>>::new();for row in csv.lines().skip(1){let c:Vec<_>=row.split(',').collect();let key=format!("{},{},{},{},{}",c[1],c[2],c[6],c[8],c[19]);groups.entry(key).or_default().entry(c[3].parse().unwrap()).or_default()[usize::from(c[0]=="after")]=c[11].parse().unwrap();}
let mut out=String::from("create_profile,passes,compressed,read_profile,corpus,paired_change_percent,bootstrap_low,bootstrap_high\n");
for(key,runs)in groups{let mut changes:Vec<_>=runs.values().map(|r|{assert!(r[0]>0.0&&r[1]>0.0);100.0*(r[1]/r[0]-1.0)}).collect();let center=med(&mut changes);let mut state=310u64;let mut boot=Vec::new();for _ in 0..10000{let mut sample=Vec::new();for _ in 0..changes.len(){state^=state<<13;state^=state>>7;state^=state<<17;sample.push(changes[state as usize%changes.len()]);}boot.push(med(&mut sample));}boot.sort_by(f64::total_cmp);out.push_str(&format!("{key},{center:.3},{:.3},{:.3}\n",boot[249],boot[9749]));}fs::write(format!("revault_lockbox_api/benches/results/issue310-shared-window-{name}-paired.csv"),&out).unwrap();println!("{name}:\n{out}");}}
