use std::{fs,process::Command};
fn main(){let mut csv=String::new();for run in 1..=4{let mut phases=[("before","target/issue310-raw-block-cache-stress-before"),("after","target/issue310-block-lru-after")];if run%2==0{phases.reverse();}for(phase,bin)in phases{let r=Command::new("taskset").args(["-c","2",bin]).env("REVAULT_ZIP_READ_CASE","small").env("REVAULT_ZIP_READ_FILES","65536").env("REVAULT_ZIP_READ_CREATE_PROFILE","BulkImport").env("REVAULT_ZIP_READ_CORPUS","pattern").env("REVAULT_ZIP_READ_MODE","plain").env("REVAULT_ZIP_READ_ACCESS","random").env("REVAULT_ZIP_READ_COMPRESSION","false").env("REVAULT_ZIP_READ_PROFILE","Interactive").env("REVAULT_ZIP_READ_PASSES","1").env("REVAULT_ZIP_READ_SAMPLES","3").env_remove("REVAULT_ZIP_READ_WRITE_PROFILE").output().unwrap();assert!(r.status.success(),"{}",String::from_utf8_lossy(&r.stderr));let text=String::from_utf8(r.stdout).unwrap();let mut lines=text.lines();let header=lines.next().unwrap();if csv.is_empty(){csv.push_str(&format!("phase,run,{header}\n"));}let row=lines.next().unwrap();assert!(lines.next().is_none());assert_eq!(row.split(',').next().unwrap(),"65536");csv.push_str(&format!("{phase},{run},{row}\n"));fs::write("revault_lockbox_api/benches/results/issue310-block-lru-stress.csv",&csv).unwrap();println!("{phase},{run},{row}");}}}
