#! /usr/bin/env dcli
import 'dart:convert';
import 'dart:io';
import 'package:dcli/dcli.dart';

const worktree = '/home/bsutton/git/.codex.workspaces/revault-issue-310-zip-read-performance';
const raw = '$worktree/docs/evidence/protected-write-current-2026-10-12/raw/build';
const cli = '$worktree/rust/target/protected-write-current-20261012/bin/lockbox-current';
const evidenceRoot = '$worktree/docs/evidence/protected-write-current-2026-10-12';
const ownedRoot = '/home/bsutton/.cache/rv-pw-20261012';
final inputPath = '${File(Platform.script.toFilePath()).parent.path}/build-inputs.json';

void require(bool ok, String message) {
  if (!ok) throw StateError(message);
}

String sha(String path) {
  final r = Process.runSync('/usr/bin/sha256sum', ['--', path]);
  require(r.exitCode == 0, 'Could not hash $path');
  return (r.stdout as String).split(RegExp(r'\s+')).first;
}

String read(String name) => File('$raw/$name').readAsStringSync();

void writeNew(String path, String text) {
  require(!File(path).existsSync(), 'Refusing to overwrite $path');
  File(path).writeAsStringSync(text, flush: true);
}

void main() {
  require(!File(inputPath).existsSync(), 'build-inputs.json already exists');
  require(read('cargo-build.exit').trim() == '0', 'Cargo build did not pass');
  final head = read('head.pre.txt').trim();
  require(head == read('head.post.txt').trim(), 'HEAD changed during the build');
  final pre = read('runtime-source-manifest.pre-rust.sha256');
  final post = read('runtime-source-manifest.post-rust.sha256');
  require(pre == post, 'Tracked Rust/Cargo source changed during build');
  final rows = <String, String>{};
  for (final line in pre.split('\n').where((line) => line.isNotEmpty)) {
    final m = RegExp(r'^([0-9a-f]{64})  (rust/.+)$').firstMatch(line);
    require(m != null, 'Malformed Rust source manifest row');
    final path = '$worktree/${m!.group(2)}';
    final digest = m.group(1)!;
    require(sha(path) == digest, 'Source changed since pre-build capture: $path');
    rows[path] = digest;
  }
  require(rows.length > 700, 'Full runtime source manifest is required');
  require(File('$worktree/rust/target/release/lockbox').existsSync(),
      'Cargo CLI executable missing');
  require(sha('$worktree/rust/target/release/lockbox') == sha(cli),
      'Preserved CLI differs from fresh Cargo output');
  final buildCommand = '''cargo +1.88.0 build --locked --offline --release -p revault_cli
cwd=$worktree/rust
TMPDIR=/home/bsutton/.cache/rv4-security-20261011
CARGO_TARGET_DIR=$worktree/rust/target
''';
  writeNew('$raw/cargo-build.command.txt', buildCommand);
  final provenancePaths = <String>[
    '$raw/cargo-build.command.txt',
    '$raw/cargo-build.stdout.log',
    '$raw/cargo-build.stderr.log',
    '$raw/cargo-build.exit',
    '$raw/head.pre.txt',
    '$raw/head.post.txt',
    '$raw/runtime-source-manifest.pre-rust.sha256',
    '$raw/runtime-source-manifest.post-rust.sha256',
    '$raw/runtime-source-diff.patch',
    '$raw/cli-artifact-hashes.sha256',
    '$raw/build.started-utc.txt',
    '$raw/build.finished-utc.txt',
    '$worktree/rust/target/release/lockbox',
    cli,
  ];
  final provenance = {for (final path in provenancePaths) path: sha(path)};
  final additionalRoots = <String>[
    '$worktree/docs/evidence/secure-arena-reclamation-2026-10-12',
    '$worktree/docs/evidence/retained-page-integration-2026-10-12',
    '$worktree/docs/evidence/native-public-extraction-2026-10-11',
    '$worktree/rust/target/release',
    '$worktree/rust/target/secure-arena-reclamation-20261012',
    '/home/bsutton/.cache/rv4-security-20261011',
    '/tmp/revault-protected-write-baseline-20261012',
    '$worktree/tools/protected-write-baseline-draft',
  ];
  final configuration = {
    'execution_authorized': true,
    'worktree': worktree,
    'head': head,
    'cli_path': cli,
    'cli_sha256': sha(cli),
    'evidence_root': evidenceRoot,
    'owned_root': ownedRoot,
    'runtime_source_hashes': rows,
    'runtime_source_manifest_sha256': sha('$raw/runtime-source-manifest.pre-rust.sha256'),
    'build_source_manifest_verified': true,
    'build_provenance_hashes': provenance,
    'additional_owned_roots': additionalRoots,
  };
  writeNew(inputPath, '${const JsonEncoder.withIndent('  ').convert(configuration)}\n');
  stdout.writeln(green(
      'Pinned current CLI and ${rows.length} Rust/Cargo source files at $head; input=${sha(inputPath)}'));
}
