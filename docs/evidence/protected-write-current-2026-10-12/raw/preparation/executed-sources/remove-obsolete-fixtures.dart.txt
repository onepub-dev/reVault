#! /usr/bin/env dcli
import 'dart:io';

const root = '/home/bsutton/.cache/revault-native-public-20261011';
const worktree = '/home/bsutton/git/.codex.workspaces/revault-issue-310-zip-read-performance';
final inventory = File('$worktree/docs/evidence/protected-write-current-2026-10-12/raw/preparation/removable-fixtures-inventory-20261012.txt');
String sha(String path) {
  final result = Process.runSync('/usr/bin/sha256sum', ['--', path]);
  if (result.exitCode != 0) throw StateError('sha256sum failed: $path');
  return (result.stdout as String).split(RegExp(r'\s+')).first;
}

void main() {
  var removedBytes = 0;
  var removed = 0;
  for (final line in inventory.readAsLinesSync()) {
    final fields = line.split('\t');
    if (fields.first != 'SOURCE' && fields.first != 'EXTRACTED_TRIAL') continue;
    if (fields.length != 5 || !fields[3].startsWith('$root/')) throw StateError('Unexpected inventory row');
    final expectedBytes = int.parse(fields[1]);
    final expectedHash = fields[2];
    final candidate = fields[3];
    final retained = fields[4].replaceFirst('retained=', '');
    if (!retained.startsWith('$root/fixtures-v2/')) throw StateError('Candidate has no retained v2 counterpart');
    final file = File(candidate);
    if (!file.existsSync() || file.lengthSync() != expectedBytes || sha(candidate) != expectedHash || sha(retained) != expectedHash) {
      throw StateError('Removal precondition changed; preserving: $candidate');
    }
    file.deleteSync();
    removedBytes += expectedBytes;
    removed++;
    stdout.writeln('REMOVED\t$expectedBytes\t$expectedHash\t$candidate\tretained=$retained');
  }
  if (removed != 17 || removedBytes != 10200547328) throw StateError('Unexpected exact removal set: $removed files / $removedBytes bytes');
  stdout.writeln('REMOVED_FILES=$removed');
  stdout.writeln('REMOVED_BYTES=$removedBytes');
}
