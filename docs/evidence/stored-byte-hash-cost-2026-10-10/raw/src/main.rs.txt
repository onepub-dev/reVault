//! Isolated algorithm-cost experiment; does not read or write Lockbox archives.
use sha2::{Digest, Sha256};
use std::{hint::black_box, time::Instant};
const SHA_DOMAIN: &[u8] = b"lockbox-v1-public-checksum/sha256";
const BLAKE_DOMAIN: &[u8] = b"revault-v4-experiment/stored-bytes/blake3";
fn bound_hash(algorithm: &str, bytes: &[u8]) -> [u8;32] {
    if algorithm == "sha256" {
        let mut h = Sha256::new();
        h.update(SHA_DOMAIN); h.update((bytes.len() as u64).to_le_bytes()); h.update(bytes);
        h.finalize().into()
    } else {
        let mut h = blake3::Hasher::new();
        h.update(BLAKE_DOMAIN); h.update(&(bytes.len() as u64).to_le_bytes()); h.update(bytes);
        *h.finalize().as_bytes()
    }
}
fn main() {
    let args: Vec<_> = std::env::args().skip(1).collect();
    assert_eq!(args.len(),3);
    let algorithm=&args[0];
    assert!(matches!(algorithm.as_str(),"sha256"|"blake3"));
    let unit: usize=args[1].parse().unwrap();
    assert!([4096,65536,262144].contains(&unit));
    let input=std::fs::read(&args[2]).unwrap();
    assert_eq!(input.len(),64*1024*1024);
    assert_eq!(blake3::hash(b"").to_hex().as_str(),"af1349b9f5f9a1a6a0404dea36dcc9499bcb25c9adc112b7cc9a93cae41f3262");
    let started=Instant::now();
    let mut aggregate=[0u8;32];
    for bytes in input.chunks(unit) {
        let digest=black_box(bound_hash(algorithm,black_box(bytes)));
        for (out,value) in aggregate.iter_mut().zip(digest) {*out^=value;}
    }
    let elapsed=started.elapsed().as_secs_f64();
    let mut expected=[0u8;32];
    for bytes in input.chunks(unit) {
        let domain=if algorithm=="sha256" {SHA_DOMAIN} else {BLAKE_DOMAIN};
        let mut message=domain.to_vec();
        message.extend_from_slice(&(bytes.len() as u64).to_le_bytes()); message.extend_from_slice(bytes);
        let digest: [u8;32]=if algorithm=="sha256" {Sha256::digest(&message).into()} else {*blake3::hash(&message).as_bytes()};
        for (out,value) in expected.iter_mut().zip(digest) {*out^=value;}
    }
    assert_eq!(aggregate,expected);
    println!("{}",serde_json::json!({"algorithm":algorithm,"unit":unit,"bytes":input.len(),"seconds":elapsed,"aggregate":aggregate,"verified":true,"scope":"single-thread domain-and-length-bound stored-byte hash cost, no archive integration"}));
}
