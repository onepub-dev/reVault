#! /usr/bin/env dcli
import 'dart:convert';
import 'dart:io';
import 'dart:math' as math;
import 'package:dcli/dcli.dart';

const variants = ['zip', 'typed-tree-1', 'typed-tree-2', 'typed-tree-4'];
const expectedCases = ['raw-random-10000x64k', 'raw-random-1gib'];
const bootstrapReplicates = 10000;
const bootstrapBaseSeed = 0x6b2175d9;
const stableKeys = [
  'source_revision', 'worktree_status', 'source_hashes', 'runner_sha256',
  'tree_test_executable', 'tree_test_executable_sha256',
  'zip_runner_executable', 'zip_runner_executable_sha256',
];

void require(bool condition, String message) {
  if (!condition) throw StateError(message);
}
Map<String, dynamic> object(Object? value, String label) {
  require(value is Map<String, dynamic>, '$label must be an object');
  return value as Map<String, dynamic>;
}
Map<String, dynamic> readObject(String path) => object(jsonDecode(File(path).readAsStringSync()), path);
int integer(Object? value, String label) {
  require(value is int, '$label must be an integer');
  return value as int;
}
double number(Object? value, String label, {bool positive = true}) {
  require(value is num, '$label must be numeric');
  final result = (value as num).toDouble();
  require(result.isFinite && (positive ? result > 0 : result >= 0), '$label is out of range');
  return result;
}
Object? canonical(Object? value) {
  if (value is Map) {
    final keys = value.keys.cast<String>().toList()..sort();
    return {for (final key in keys) key: canonical(value[key])};
  }
  if (value is List) return value.map(canonical).toList();
  return value;
}
bool same(Object? left, Object? right) => jsonEncode(canonical(left)) == jsonEncode(canonical(right));
void hashValue(Object? value, String label) => require(
    value is String && RegExp(r'^[0-9a-f]{64}$').hasMatch(value), '$label is not a SHA-256 identity');
String sha(String path) {
  final result = Process.runSync('sha256sum', ['--', path]);
  require(result.exitCode == 0, 'Cannot hash evidence file $path');
  final digest = '${result.stdout}'.split(RegExp(r'\s+')).first;
  hashValue(digest, path);
  return digest;
}

class Observation {
  Observation(this.round, this.elapsed, this.cpu, this.throughput, this.rss,
      this.processRss, this.readBytes, this.writeBytes, this.cancelledBytes, this.payload);
  final int round;
  final double elapsed, cpu, throughput, rss, processRss;
  final int readBytes, writeBytes, cancelledBytes, payload;
}

Observation validateSample(Map<String, dynamic> row, String caseName, String variant,
    int totalRounds, int warmups, int files, Map<String, dynamic> identity, String affinity) {
  final round = integer(row['round'], '$caseName/$variant round');
  require(round >= 0 && round < totalRounds, '$caseName/$variant unexpected round $round');
  require(row['case'] == caseName && row['backend'] == variant, 'Sample identity mismatch');
  require(row['warmup'] == (round < warmups), 'Warmup flag does not match planned round');
  require(row['verified'] == true && row['exit_code'] == 0 &&
      row['source_and_output_byte_verification'] == true, 'Unverified sample $caseName/$variant/$round');
  require(row['durable'] == true && row['cold_disk_gate_passed'] == true, 'Sample is not cold/durable');
  require(row['affinity'] == affinity, 'Unexpected CPU affinity');
  require(row['executable_stable_for_sample'] == true, 'Executable changed during sample');
  final isTree = variant != 'zip';
  final expectedExecutable = identity[isTree ? 'tree_test_executable' : 'zip_runner_executable'];
  final expectedHash = identity[isTree ? 'tree_test_executable_sha256' : 'zip_runner_executable_sha256'];
  require(row['executable'] == expectedExecutable && row['executable_sha256_before'] == expectedHash &&
      row['executable_sha256_after'] == expectedHash, 'Executable differs from batch identity');
  final gate = object(row['cache_gate'], 'cache gate');
  require(gate['success'] == true && gate['resident_bytes_after_dontneed'] == 0 &&
      gate['required_resident_bytes'] == 0 && gate['sync_exit_code'] == 0 &&
      gate['fadvise_exit_code'] == 0 && gate['fincore_exit_code'] == 0, 'Invalid archive eviction/residency gate');
  final sample = object(row['sample'], 'sample');
  require(sample['kind'] == 'filesystem_extraction' && sample['verified'] == true &&
      sample['files'] == files && sample['file_syncs'] == files && sample['directory_syncs'] == 2,
      'Extraction verification/sync scope mismatch');
  require(sample['backend'] == (isTree ? 'typed-tree' : 'zip'), 'Unexpected extraction backend');
  require(sample['durability'] is String && (sample['durability'] as String).contains('sync_all'),
      'Durable extraction scope is not reported');
  final environment = object(row['environment'], 'sample environment');
  require(environment['REVAULT_EXTRACT_DURABLE'] == '1', 'Durability was not explicitly requested');
  if (isTree) {
    final workers = int.parse(variant.split('-').last);
    require(row['worker_count'] == workers && sample['configured_cpu_workers'] == workers &&
        sample['worker_threads'] == (workers == 1 ? 0 : workers) &&
        sample['max_queued_chunks'] == (workers == 1 ? 1 : workers * 8) &&
        sample['single_chunk_serial_fallback'] == true && sample['parallel_min_requested_bytes'] == 131072 &&
        environment['REVAULT_READ_WORKERS'] == '$workers', 'Worker policy mismatch');
  } else {
    require(row['worker_count'] == null, 'ZIP baseline must remain serial');
  }
  final payload = integer(sample['logical_bytes_written'], 'payload bytes');
  require(payload > 0 && row['output_total_bytes'] == payload && row['output_file_count'] == files,
      'Output inventory mismatch');
  final io = object(sample['storage_io'], 'storage I/O');
  final reads = integer(io['read_bytes'], 'storage reads');
  final writes = integer(io['write_bytes'], 'storage writes');
  final cancelled = integer(io['cancelled_write_bytes'], 'cancelled writes');
  final minimum = (payload * 9 + 9) ~/ 10;
  require(reads >= minimum && writes >= 0 && cancelled >= 0, 'Insufficient storage traffic or invalid counters');
  require(row['observed_process_storage_read_bytes'] == reads &&
      integer(row['minimum_storage_read_bytes'], 'read gate') >= minimum,
      'Recorded read-byte gate does not enforce 90%');
  final resources = object(sample['resources'], 'resources');
  final cpu = object(resources['cpu'], 'CPU resources');
  final elapsed = number(sample['total_seconds'], 'elapsed seconds');
  final throughput = number(sample['mib_per_second'], 'throughput');
  final calculated = payload / 1048576 / elapsed;
  require((throughput - calculated).abs() <= calculated * 1e-9, 'Inconsistent throughput');
  return Observation(round, elapsed, number(cpu['cpu_seconds'], 'CPU seconds'), throughput,
      number(resources['peak_rss_kib'], 'extraction peak RSS'),
      number(row['process_max_rss_kib'], 'whole-process peak RSS'), reads, writes, cancelled, payload);
}

double median(Iterable<double> values) {
  final sorted = values.toList()..sort();
  require(sorted.isNotEmpty, 'Empty median');
  final n = sorted.length;
  return n.isOdd ? sorted[n ~/ 2] : (sorted[n ~/ 2 - 1] + sorted[n ~/ 2]) / 2;
}
double percentile(List<double> sorted, double fraction) {
  final position = fraction * (sorted.length - 1);
  final lower = position.floor();
  final upper = position.ceil();
  return sorted[lower] + (sorted[upper] - sorted[lower]) * (position - lower);
}
class Seeded {
  Seeded(this.state) { require(state != 0, 'Zero PRNG seed'); }
  int state;
  int nextInt(int bound) {
    final limit = 0x100000000 - (0x100000000 % bound);
    while (true) {
      var x = state;
      x ^= (x << 13) & 0xffffffff;
      x ^= x >> 17;
      x ^= (x << 5) & 0xffffffff;
      state = x & 0xffffffff;
      if (state < limit) return state % bound;
    }
  }
}
int seedFor(String identity) {
  var value = bootstrapBaseSeed;
  for (final byte in utf8.encode(identity)) { value = ((value ^ byte) * 16777619) & 0xffffffff; }
  return value == 0 ? 1 : value;
}
Map<String, dynamic> paired(String label, Map<int, Observation> candidate,
    Map<int, Observation> baseline, double Function(Observation) metric) {
  final rounds = candidate.keys.toList()..sort();
  require(same(rounds, baseline.keys.toList()..sort()), 'Unpaired comparison $label');
  final logs = [for (final round in rounds) math.log(metric(candidate[round]!) / metric(baseline[round]!))];
  final seed = seedFor(label);
  final random = Seeded(seed);
  final bootstraps = List<double>.generate(bootstrapReplicates, (_) {
    var sum = 0.0;
    for (var n = 0; n < logs.length; n++) { sum += logs[random.nextInt(logs.length)]; }
    return math.exp(sum / logs.length);
  })..sort();
  return {
    'pairs': rounds.length, 'paired_rounds': rounds,
    'geometric_mean_ratio': math.exp(logs.reduce((a, b) => a + b) / logs.length),
    'low95': percentile(bootstraps, 0.025), 'high95': percentile(bootstraps, 0.975),
    'method': 'paired percentile bootstrap of mean log(candidate/baseline); linear percentile interpolation',
    'bootstrap_replicates': bootstrapReplicates, 'seed_xorshift32': seed,
    'interpretation': 'ratio below one favors candidate; no multiplicity correction',
  };
}
Map<String, dynamic> metrics(List<Observation> rows) => {
  'measured_samples': rows.length,
  'median_elapsed_seconds': median(rows.map((row) => row.elapsed)),
  'median_cpu_seconds': median(rows.map((row) => row.cpu)),
  'median_mib_per_second': median(rows.map((row) => row.throughput)),
  'max_extraction_peak_rss_kib': rows.map((row) => row.rss).reduce((a, b) => a > b ? a : b),
  'max_whole_process_rss_kib': rows.map((row) => row.processRss).reduce((a, b) => a > b ? a : b),
  'storage_read_bytes_range': [rows.map((row) => row.readBytes).reduce((a, b) => a < b ? a : b), rows.map((row) => row.readBytes).reduce((a, b) => a > b ? a : b)],
  'storage_read_bytes_per_payload_byte_range': [
    rows.map((row) => row.readBytes / row.payload).reduce((a, b) => a < b ? a : b),
    rows.map((row) => row.readBytes / row.payload).reduce((a, b) => a > b ? a : b)],
  'storage_write_bytes_range': [rows.map((row) => row.writeBytes).reduce((a, b) => a < b ? a : b), rows.map((row) => row.writeBytes).reduce((a, b) => a > b ? a : b)],
  'max_cancelled_write_bytes': rows.map((row) => row.cancelledBytes).reduce((a, b) => a > b ? a : b),
};

void main(List<String> args) {
  require(args.length <= 2, 'Usage: ./summarize.dart [BATCH_DIR [OUTPUT_JSON]]');
  final root = Directory(args.isEmpty ? '/tmp/revault-parallel-20261010/cold-batch' : args[0]).absolute.path;
  final output = File(args.length == 2 ? args[1] : '$root/metrics-summary.json');
  require(!output.existsSync(), 'Refusing to replace existing summary ${output.path}');
  final paths = <String>['$root/complete.json', '$root/identity-before.json', '$root/identity-after.json'];
  final complete = readObject(paths[0]);
  require(complete['complete'] == true && complete['durable'] == true && complete['zip_is_serial'] == true,
      'Batch completion/durability/serial baseline gate failed');
  final measured = integer(complete['measured_rounds_per_case'], 'planned measured rounds');
  final warmups = integer(complete['warmup_rounds_per_case'], 'planned warmup rounds');
  require(measured == 12 && warmups == 3, 'This planned batch requires 12 measured and 3 excluded warmup rounds');
  final total = measured + warmups;
  require(same(complete['variants'], variants), 'All four planned variants are required');
  final affinity = complete['affinity_mask_for_every_sample'];
  require(affinity is String && affinity.isNotEmpty, 'Missing affinity policy');
  require(complete['artifact_cap_bytes'] == 20 * 1024 * 1024 * 1024 &&
      integer(complete['apparent_bytes_after'], 'artifact bytes') <= integer(complete['artifact_cap_bytes'], 'artifact cap'),
      'Artifact budget was exceeded');
  final before = readObject(paths[1]);
  final after = readObject(paths[2]);
  for (final key in stableKeys) {
    require(before[key] != null && same(before[key], after[key]), 'Unstable batch source/executable identity: $key');
  }
  require(object(before['source_hashes'], 'source identities').isNotEmpty, 'Missing source hashes');
  for (final entry in object(before['source_hashes'], 'source identities').entries) { hashValue(entry.value, entry.key); }
  for (final key in ['runner_sha256', 'tree_test_executable_sha256', 'zip_runner_executable_sha256']) { hashValue(before[key], key); }
  final audits = (complete['cases'] as List).map((value) => object(value, 'case audit')).toList();
  final caseNames = audits.map((value) => value['case'] as String).toList()..sort();
  require(same(caseNames, expectedCases), 'Both distinct planned workloads are required');
  final results = <String, dynamic>{};
  var allRows = 0;
  for (final caseName in caseNames) {
    final auditPath = '$root/$caseName/case-audit.json';
    paths.add(auditPath);
    final audit = readObject(auditPath);
    require(same(audit, audits.singleWhere((value) => value['case'] == caseName)), 'Case audit disagrees with completion record');
    require(audit['stable'] == true && audit['failure'] == null && audit['rounds_complete'] == total &&
        audit['warmup_rounds_planned'] == warmups && audit['measured_rounds_planned'] == measured &&
        same(audit['input_identity_before'], audit['input_identity_after']), 'Unstable or incomplete case audit');
    final inputs = object(audit['input_identity_before'], 'fixture identities');
    for (final key in ['case_json_sha256', 'inventory_sha256', 'archive_sha256', 'tree_sha256', 'tree_public_sha256']) { hashValue(inputs[key], key); }
    final source = object(inputs['source'], 'source inventory identity');
    final files = integer(source['file_count'], 'source count');
    require(files > 0 && source['unique_file_hashes'] == files, 'Source files are not individually varied');
    hashValue(source['aggregate_manifest_sha256'], 'source aggregate');
    require(source['inventory_sha256'] == inputs['inventory_sha256'], 'Source inventory identity mismatch');
    final counts = object(audit['sample_counts'], 'sample counts');
    final series = <String, Map<int, Observation>>{};
    int? payload;
    for (final variant in variants) {
      require(counts[variant] == total, 'Missing planned sample count');
      final samplePath = '$root/$caseName/$variant/samples.jsonl';
      paths.add(samplePath);
      final lines = File(samplePath).readAsLinesSync().where((line) => line.trim().isNotEmpty).toList();
      require(lines.length == total, 'Expected $total rows at $samplePath');
      final byRound = <int, Observation>{};
      for (final line in lines) {
        final row = object(jsonDecode(line), samplePath);
        final observation = validateSample(row, caseName, variant, total, warmups, files, before, affinity as String);
        require(!byRound.containsKey(observation.round), 'Duplicate round in $samplePath');
        payload ??= observation.payload;
        require(payload == observation.payload, 'Logical payload changed between paired samples');
        byRound[observation.round] = observation;
      }
      require(byRound.length == total, 'Incomplete rounds');
      allRows += byRound.length;
      byRound.removeWhere((round, _) => round < warmups);
      require(byRound.length == measured, 'Wrong measured sample count after excluding warmups');
      series[variant] = byRound;
    }
    final expectedPayload = caseName == 'raw-random-1gib' ? 1073741824 : 655360000;
    final expectedFiles = caseName == 'raw-random-1gib' ? 1 : 10000;
    require(payload == expectedPayload && files == expectedFiles, 'Workload was silently resized');
    final comparisons = <String, dynamic>{};
    for (final pair in [
      ['typed-tree-1', 'zip'], ['typed-tree-2', 'zip'], ['typed-tree-4', 'zip'],
      ['typed-tree-2', 'typed-tree-1'], ['typed-tree-4', 'typed-tree-1'],
    ]) {
      final label = '${pair[0]}_vs_${pair[1]}';
      comparisons[label] = {
        'elapsed_seconds': paired('$caseName/$label/elapsed', series[pair[0]]!, series[pair[1]]!, (row) => row.elapsed),
        'cpu_seconds': paired('$caseName/$label/cpu', series[pair[0]]!, series[pair[1]]!, (row) => row.cpu),
      };
    }
    results[caseName] = {
      'files': files, 'logical_payload_bytes': payload, 'warmup_rounds_excluded': warmups,
      'measured_rounds': measured, 'variants': {for (final variant in variants) variant: metrics(series[variant]!.values.toList())},
      'paired_comparisons': comparisons,
    };
  }
  require(complete['sample_rows'] == allRows && allRows == expectedCases.length * variants.length * total,
      'Completion record/sample row total mismatch');
  final summary = {
    'complete': true, 'generated_utc': DateTime.now().toUtc().toIso8601String(),
    'scope': 'cold archive page cache plus observed process storage reads; durable regular-file extraction; not hardware-cache-cold or public CLI qualification',
    'gates': {'all_samples_verified': true, 'zero_archive_residency_before_each_sample': true,
      'storage_reads_at_least_90_percent_of_payload': true, 'stable_input_source_and_executable_audits': true,
      'file_and_both_directory_syncs_in_timer': true, 'all_four_variants': true,
      'measured_rounds_per_case': measured, 'excluded_warmup_rounds_per_case': warmups},
    'rss_scope': 'max_extraction_peak_rss is process-lifetime peak observed before verification, including startup; whole-process max additionally includes post-timing verification',
    'statistics': {'bootstrap_replicates': bootstrapReplicates, 'base_seed': bootstrapBaseSeed,
      'pairing': 'same measured round; warmups excluded; ratios candidate/baseline',
      'confidence_interval': '95% paired percentile bootstrap on mean log ratios, no multiplicity adjustment',
      'median_definition': 'mean of central two order statistics for even sample counts'},
    'metadata': {'complete_record': complete, 'identity_before': before, 'identity_after': after},
    'evidence_sha256': {for (final path in paths) path: sha(path)},
    'summarizer_sha256': sha(Platform.script.toFilePath()),
    'cases': results,
  };
  output.parent.createSync(recursive: true);
  output.writeAsStringSync('${const JsonEncoder.withIndent('  ').convert(summary)}\n', flush: true);
  print(green('Validated and summarized $allRows rows (warmups excluded from metrics): ${output.path}'));
}
