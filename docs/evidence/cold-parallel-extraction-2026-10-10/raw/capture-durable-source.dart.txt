#! /usr/bin/env dcli
import 'dart:convert';
import 'dart:io';
import 'package:dcli/dcli.dart';

const base = '/tmp/revault-parallel-20261010';
const tree = '/home/bsutton/git/.codex.workspaces/revault-issue-310-zip-read-performance';
void require(bool condition, String message) { if (!condition) throw StateError(message); }
String command(String exe, List<String> args) {
  final result = Process.runSync(exe, args, workingDirectory: tree);
  require(result.exitCode == 0, '$exe failed: ${result.stderr}');
  return '${result.stdout}';
}
String hash(String path) => command('sha256sum', ['--', path]).split(RegExp(r'\s+')).first;
void main() {
  final complete = jsonDecode(File('$base/cold-batch/complete.json').readAsStringSync()) as Map;
  final summary = jsonDecode(File('$base/cold-batch/metrics-summary.json').readAsStringSync()) as Map;
  require(complete['complete'] == true && summary['complete'] == true, 'Durable batch and summary must be complete');
  final identity = summary['metadata']['identity_after'] as Map;
  require(command('git', ['rev-parse', 'HEAD']).trim() == identity['source_revision'], 'Revision changed');
  final destination = Directory('$base/durable-source');
  require(!destination.existsSync(), 'Refusing to overwrite captured source');
  final sourceHashes = identity['source_hashes'] as Map;
  for (final entry in sourceHashes.entries) {
    final path = entry.key as String;
    require(!path.startsWith('/') && !path.split('/').contains('..'), 'Invalid source path');
    require(hash('$tree/rust/$path') == entry.value, 'Original measured source changed: $path');
  }
  destination.createSync();
  for (final entry in sourceHashes.entries) {
    final output = File('${destination.path}/rust/${entry.key}');
    output.parent.createSync(recursive: true);
    File('$tree/rust/${entry.key}').copySync(output.path);
    require(hash(output.path) == entry.value, 'Copy mismatch');
  }
  File('${destination.path}/source.patch').writeAsStringSync(command('git', ['diff', '--', 'rust']));
  File('${destination.path}/status.txt').writeAsStringSync(command('git', ['status', '--porcelain=v1']));
  File('${destination.path}/identity.json').writeAsStringSync(const JsonEncoder.withIndent('  ').convert(identity));
  print('Captured and verified ${sourceHashes.length} original measured Rust/Cargo files.');
}
