#! /usr/bin/env dcli
import 'dart:convert';
import 'dart:io';
import 'package:dcli/dcli.dart';

const out = '/tmp/revault-open-admission-20261010/diagnostic';
const fixtureBase = '/tmp/revault-selective-read-20261010/fixtures';
const binary = '/tmp/revault-open-admission-20261010/bin/diagnostic';
const rust = '/home/bsutton/git/.codex.workspaces/revault-issue-310-zip-read-performance/rust';
const fixtureNames = ['raw8m-random', 'raw64m-random', 'compressed8m-pattern', 'small512-mixed-compressed'];
const repeats = 100;

String sha(String path) {
  final result = Process.runSync('sha256sum', [path]);
  if (result.exitCode != 0) throw StateError('sha256sum $path failed: ${result.stderr}');
  return '${result.stdout}'.split(RegExp(r'\s+')).first;
}

void hashTree(Directory dir, String relative, Map<String, String> output) {
  for (final entity in dir.listSync(followLinks: true)) {
    final name = relative.isEmpty ? entity.uri.pathSegments.last : '$relative/${entity.uri.pathSegments.last}';
    final type = FileSystemEntity.typeSync(entity.path, followLinks: true);
    if (type == FileSystemEntityType.directory) {
      hashTree(Directory(entity.path), name, output);
    } else if (type == FileSystemEntityType.file) {
      output[name] = sha(entity.path);
    }
  }
}

Map<String, String> fixtureHashes(String name) {
  final root = Directory('$fixtureBase/$name');
  final hashes = <String, String>{};
  hashTree(root, '', hashes);
  if (hashes.isEmpty) throw StateError('Fixture hash inventory unexpectedly empty for $name');
  return hashes;
}

double median(List<double> values) {
  final sorted = [...values]..sort();
  if (sorted.isEmpty) throw StateError('Cannot summarize empty values');
  final mid = sorted.length ~/ 2;
  return sorted.length.isOdd ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}

Map<String, dynamic> summarize(List<Map<String, dynamic>> observations) {
  final stageNames = <String>{};
  for (final row in observations) stageNames.addAll((row['stages_seconds'] as Map<String, dynamic>).keys);
  final stages = <String, dynamic>{};
  for (final stage in stageNames) {
    final stageSeconds = observations.map((row) => ((row['stages_seconds'] as Map<String, dynamic>)[stage] as num).toDouble()).toList();
    final functionNames = <String>{};
    for (final row in observations) {
      final byStage = row['functions_by_stage'] as Map<String, dynamic>;
      final functions = byStage[stage];
      if (functions is Map<String, dynamic>) functionNames.addAll(functions.keys);
    }
    final functions = <String, dynamic>{};
    for (final function in functionNames) {
      final entries = <Map<String, dynamic>>[];
      for (final row in observations) {
        final byStage = row['functions_by_stage'] as Map<String, dynamic>;
        final stageFunctions = byStage[stage];
        if (stageFunctions is Map<String, dynamic> && stageFunctions[function] is Map<String, dynamic>) {
          entries.add(stageFunctions[function] as Map<String, dynamic>);
        }
      }
      functions[function] = {
        'observations': entries.length,
        'median_calls': median(entries.map((e) => (e['calls'] as num).toDouble()).toList()),
        'median_failures': median(entries.map((e) => (e['failures'] as num).toDouble()).toList()),
        'median_seconds': median(entries.map((e) => (e['seconds'] as num).toDouble()).toList()),
      };
    }
    final storageNames = <String>{};
    for (final row in observations) {
      final byStage = row['storage_by_stage'] as Map<String, dynamic>;
      final storage = byStage[stage];
      if (storage is Map<String, dynamic>) storageNames.addAll(storage.keys);
    }
    final storage = <String, dynamic>{};
    for (final name in storageNames) {
      final values = <double>[];
      for (final row in observations) {
        final data = (row['storage_by_stage'] as Map<String, dynamic>)[stage];
        if (data is Map<String, dynamic> && data[name] is num) values.add((data[name] as num).toDouble());
      }
      if (values.isNotEmpty) storage[name] = median(values);
    }
    stages[stage] = {
      'median_stage_seconds': median(stageSeconds),
      'median_function_timers': functions,
      'median_storage_counts_and_seconds': storage,
    };
  }
  return {
    'observation_count': observations.length,
    'median_total_seconds': median(observations.map((row) => (row['total_seconds'] as num).toDouble()).toList()),
    'stages': stages,
  };
}

void main() {
  Directory(out).createSync(recursive: true);
  final sourcePaths = [
    '$rust/Cargo.lock',
    '$rust/revault_lockbox_api/src/file_format/mod.rs',
    '$rust/revault_lockbox_api/src/file_format/diagnostic_timing.rs',
    '$rust/revault_lockbox_api/src/file_format/authenticated_index/cache.rs',
    '$rust/revault_lockbox_api/src/file_format/candidate_files/resource_probe/tree_probe.rs',
    '$rust/revault_lockbox_api/src/file_format/candidate_files/tree_image/reader.rs',
    '$rust/revault_lockbox_api/src/file_format/publication_anchor/shared.rs',
    Platform.script.toFilePath(),
  ];
  final beforeFixtures = {for (final name in fixtureNames) name: fixtureHashes(name)};
  final beforeSources = {for (final path in sourcePaths) path: sha(path)};
  final beforeBinary = sha(binary);
  final summaries = <String, dynamic>{};
  final raw = <String, dynamic>{};
  for (final name in fixtureNames) {
    final fixture = '$fixtureBase/$name/other';
    final unit = name == 'compressed8m-pattern' || name == 'small512-mixed-compressed' ? 262144 : 65536;
    final env = {...Platform.environment,
      'REVAULT_CANDIDATE_PHASE': 'tree-sample',
      'REVAULT_CANDIDATE_ROOT': fixture,
      'REVAULT_TREE_OPEN_DIAGNOSTIC': '$repeats',
      'REVAULT_CANDIDATE_ACCESS': 'range',
      'REVAULT_CANDIDATE_UNIT': '$unit',
      'REVAULT_CANDIDATE_PASSES': '1',
    };
    final result = Process.runSync('taskset', [
      '-c', '2', binary, '--ignored', '--exact',
      'file_format::candidate_files::resource_probe::candidate_file_resource_probe',
      '--nocapture', '--test-threads=1',
    ], environment: env, workingDirectory: rust);
    final output = '${result.stdout}${result.stderr}';
    File('$out/logs-$name.log').writeAsStringSync(output, flush: true);
    if (result.exitCode != 0) throw StateError('$name diagnostic failed (${result.exitCode}); see $out/logs-$name.log');
    const marker = 'TREE_OPEN_DIAGNOSTIC ';
    final markerLine = output.split('\n').where((line) => line.contains(marker)).toList();
    if (markerLine.length != 1) throw StateError('$name has ${markerLine.length} diagnostic markers');
    final markerAt = markerLine.single.indexOf(marker);
    final payload = jsonDecode(markerLine.single.substring(markerAt + marker.length)) as Map<String, dynamic>;
    final observations = (payload['observations'] as List).cast<Map<String, dynamic>>();
    if (payload['verified'] != true || payload['repeats'] != repeats || observations.length != repeats) {
      throw StateError('$name verification/repeat count mismatch');
    }
    summaries[name] = summarize(observations);
    raw[name] = payload;
    print('DIAGNOSTIC_OK $name observations=${observations.length} verified=${payload['verified']} archive=${payload['archive_sha256']}');
  }
  final afterFixtures = {for (final name in fixtureNames) name: fixtureHashes(name)};
  final afterSources = {for (final path in sourcePaths) path: sha(path)};
  final afterBinary = sha(binary);
  final stable = beforeBinary == afterBinary && jsonEncode(beforeSources) == jsonEncode(afterSources)
      && jsonEncode(beforeFixtures) == jsonEncode(afterFixtures);
  final audit = {
    'scope': '100 fresh observed opens per fixture; verification before/after each diagnostic process; CPU2; instrumented only; no ZIP comparison',
    'source_revision': Process.runSync('git', ['rev-parse', 'HEAD'], workingDirectory: rust).stdout.toString().trim(),
    'binary_sha256_before_after': [beforeBinary, afterBinary],
    'source_sha256_before': beforeSources,
    'source_sha256_after': afterSources,
    'fixtures_sha256_before_follow_links': beforeFixtures,
    'fixtures_sha256_after_follow_links': afterFixtures,
    'fixture_hashes_nonempty': beforeFixtures.values.every((m) => m.isNotEmpty) && afterFixtures.values.every((m) => m.isNotEmpty),
    'stable': stable,
    'cpu_affinity': 2,
    'summary': summaries,
    'raw_payloads': raw,
  };
  File('$out/summary.json').writeAsStringSync(const JsonEncoder.withIndent('  ').convert(audit), flush: true);
  if (!stable) throw StateError('source, binary, or fixture hashes changed');
  print('AUDIT_OK $out/summary.json');
}
