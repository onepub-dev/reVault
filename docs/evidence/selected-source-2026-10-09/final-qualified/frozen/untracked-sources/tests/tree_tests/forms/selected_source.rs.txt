pub(super) fn prepare_texts(storage:&impl Storage,mode:FormatMode,authority:&Authority<'_>)->Result<(Vec<crate::file_format::authenticated_index::Entry>,tree::PreparedStoragePayloadPlan<'static>)>{
 let (mut catalogue,base,content_key)={let mut opened=TreeImage::open(crate::file_format::allocation_map::compaction::View(storage),archive(),mode,authority,key(mode))?;opened.image.verify_all()?;(opened.image.catalogue,shared::commitment(&opened.image.anchor)?,opened.image.value_key.ok_or(Error::CorruptRecord)?)};
 let old:Vec<_>=catalogue.forms.texts().into_iter().cloned().collect();let retired:Vec<_>=old.iter().flat_map(|l|l.extents.iter().copied()).collect();
 let source=tree::SelectedSource::open(storage,archive(),mode,authority,key(mode),base,retired.clone())?;
 let mut pages=PreparedSecurePages::new(4096)?;let mut callbacks=Vec::new();let mut ids=Vec::new();let mut layouts=Vec::new();
 for old in &old{let prepared=form_segments::prepare_stored(old,source.view(storage),archive(),mode,&content_key,old.id,old.context,old.revision.checked_add(1).ok_or(Error::CorruptRecord)?,old.sensitivity)?;let count=prepared.pages.len();pages.append(prepared.pages)?;callbacks.push((count,prepared.payload));ids.push((old.id,count));layouts.push(prepared.layout);}
 for (target,new)in catalogue.forms.texts_mut().into_iter().zip(layouts){*target=new;}
 let records=catalogue.tree_records()?;
 let plan=tree::PreparedStoragePayloadPlan{source,retired,pages,payload:Box::new(move|mut n,view|{for(count,callback)in &mut callbacks{if n<*count{return callback(n,view);}n-=*count;}Err(Error::CorruptRecord)}),rebind:Box::new(move|extents|{let mut offset=0;let mut places=BTreeMap::new();for(id,count)in ids{let end=offset+count;places.insert(id,extents.get(offset..end).ok_or(Error::CorruptRecord)?);offset=end;}if offset!=extents.len(){return Err(Error::CorruptRecord);}for layout in catalogue.forms.texts_mut(){layout.rebind(places.remove(&layout.id).ok_or(Error::CorruptRecord)?)?;}if !places.is_empty(){return Err(Error::CorruptRecord);}catalogue.tree_records()})};
 Ok((records,plan))
}
pub(super) fn replace_texts(storage:&mut impl Storage,mode:FormatMode,authority:&Authority<'_>,owner:&OwnerSigningKeyPair)->Result<bool>{
 let (records,plan)=prepare_texts(&*storage,mode,authority)?;
 tree::rewrite_prepared_storage_payload_records(storage,archive(),mode,authority,mode.signed().then_some(owner),key(mode),records,plan)
}
#[derive(Debug)]
struct CloneTrap<S>{inner:Guarded<S>,reads:std::cell::Cell<usize>,writes:std::cell::Cell<usize>,mutations:std::cell::Cell<usize>}
impl<S> Clone for CloneTrap<S>{fn clone(&self)->Self{panic!("selected-source path cloned underlying storage")}}
impl<S:Storage> CloneTrap<S>{fn observe(&self,at:u64,len:usize){if self.inner.spans.borrow().iter().any(|(a,b)|at<*b&&*a<at+len as u64){self.reads.set(self.reads.get()+1);}}}
impl<S:Storage> Storage for CloneTrap<S>{
 fn len(&self)->Result<u64>{self.inner.len()}
 fn read_at(&self,at:u64,len:usize)->Result<Vec<u8>>{self.observe(at,len);self.inner.read_at(at,len)}
 fn read_at_into(&self,at:u64,out:&mut[u8])->Result<()>{self.observe(at,out.len());self.inner.read_at_into(at,out)}
 fn append(&mut self,bytes:&[u8])->Result<u64>{self.mutations.set(self.mutations.get()+1);if bytes.starts_with(crate::file_format::page::PAGE_MAGIC){self.writes.set(self.writes.get()+1);}self.inner.append(bytes)}
 fn write_at(&mut self,at:u64,bytes:&[u8])->Result<()>{self.mutations.set(self.mutations.get()+1);if bytes.starts_with(crate::file_format::page::PAGE_MAGIC){self.writes.set(self.writes.get()+1);}self.inner.write_at(at,bytes)}
 fn truncate(&mut self,len:u64)->Result<()>{self.mutations.set(self.mutations.get()+1);self.inner.truncate(len)}
 fn sync(&self)->Result<()>{self.mutations.set(self.mutations.get()+1);self.inner.sync()}
}
