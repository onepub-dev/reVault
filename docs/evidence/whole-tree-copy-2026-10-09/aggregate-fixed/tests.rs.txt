#[derive(Debug)]
struct CopyGuard<S>{inner:S,all:Vec<Extent>,phase:Rc<Cell<bool>>,destination:bool,mutations:Rc<Cell<usize>>,copy_read:Rc<Cell<u64>>,published:Rc<Cell<bool>>}
impl<S> Clone for CopyGuard<S>{fn clone(&self)->Self{panic!("compaction must not clone source or destination")}}
impl<S:Storage> Storage for CopyGuard<S>{
 fn len(&self)->Result<u64>{self.inner.len()}
 fn read_at(&self,at:u64,len:usize)->Result<Vec<u8>>{
  assert!(!self.phase.get()||!self.all.iter().any(|e|at<e.start+e.len&&e.start<at+len as u64),"ordinary payload read during relocation/readback");self.inner.read_at(at,len)
 }
 fn read_at_into(&self,at:u64,out:&mut[u8])->Result<()>{if self.phase.get()&&self.all.iter().any(|e|at<e.start+e.len&&e.start<at+out.len() as u64){self.copy_read.set(self.copy_read.get()+out.len() as u64);}self.inner.read_at_into(at,out)}
 fn append(&mut self,b:&[u8])->Result<u64>{assert!(self.destination,"source mutation");self.mutations.set(self.mutations.get()+1);self.phase.set(true);self.inner.append(b)}
 fn write_at(&mut self,at:u64,b:&[u8])->Result<()>{assert!(self.destination,"source mutation");self.mutations.set(self.mutations.get()+1);if at==0&&b.starts_with(b"RV4SHR01"){self.phase.set(false);self.published.set(true);}self.inner.write_at(at,b)}
 fn truncate(&mut self,n:u64)->Result<()>{assert!(self.destination,"source mutation");self.mutations.set(self.mutations.get()+1);self.inner.truncate(n)}
 fn sync(&self)->Result<()>{assert!(self.destination,"source mutation");self.mutations.set(self.mutations.get()+1);self.inner.sync()}
}
fn hash(storage:&impl Storage)->[u8;32]{let mut h=Sha256::new();let len=storage.len().unwrap();h.update(len.to_le_bytes());let mut at=0;while at<len{let n=(len-at).min(65536) as usize;storage.read_at_secure(at,n).unwrap().with_bytes(|b|h.update(b)).unwrap();at+=n as u64;}h.finalize().into()}
fn secure_spans(c:&super::super::super::super::dense_catalogue::Catalogue)->Vec<(u64,u64)>{c.variables.iter().flat_map(|v|v.layout.extents.iter()).chain(c.forms.texts().into_iter().flat_map(|t|t.extents.iter())).map(|e|(e.start,e.start+e.len)).collect()}
fn rows(c:&super::super::super::super::dense_catalogue::Catalogue)->Vec<(u8,Vec<u8>,Vec<u8>)>{c.tree_records().unwrap().into_iter().map(|r|(r.namespace,r.key.to_vec(),r.value.to_vec())).collect()}

  // Retain the previous one-shot construction independently of the factored helper.
  let mut original=Sha256::new();original.update(b"lockbox-v1-public-checksum/sha256");original.update((length as u64).to_le_bytes());original.update(&bytes);let expected:[u8;32]=original.finalize().into();

#[derive(Clone,Debug)]
struct SharedStore(Rc<std::cell::RefCell<StorageBackend>>);
impl SharedStore{fn new(bytes:Vec<u8>)->Self{Self(Rc::new(std::cell::RefCell::new(StorageBackend::memory(bytes))))}}
impl Storage for SharedStore{
 fn len(&self)->Result<u64>{self.0.borrow().len()}
 fn read_at(&self,at:u64,len:usize)->Result<Vec<u8>>{self.0.borrow().read_at(at,len)}
 fn read_at_into(&self,at:u64,out:&mut[u8])->Result<()>{self.0.borrow().read_at_into(at,out)}
 fn append(&mut self,b:&[u8])->Result<u64>{self.0.borrow_mut().append(b)}
 fn write_at(&mut self,at:u64,b:&[u8])->Result<()>{self.0.borrow_mut().write_at(at,b)}
 fn truncate(&mut self,n:u64)->Result<()>{self.0.borrow_mut().truncate(n)}
 fn sync(&self)->Result<()>{self.0.borrow().sync()}
}

 (destination,output,mutations)
}

#[derive(Debug)]
struct ReadbackFailure {
 inner:Guarded<crate::file_format::preparation_journal::tests::CrashStore>,
 trigger:u64,tamper:bool,triggered:Cell<bool>,cleanup:Option<u8>,truncated:Cell<bool>,
}
impl Clone for ReadbackFailure{fn clone(&self)->Self{panic!("copy must not clone destination")}}
impl Storage for ReadbackFailure{
 fn len(&self)->Result<u64>{self.inner.len()}
 fn read_at(&self,at:u64,len:usize)->Result<Vec<u8>>{self.inner.read_at(at,len)}
 fn read_at_into(&self,at:u64,out:&mut[u8])->Result<()>{
  self.inner.read_at_into(at,out)?;
  if at==self.trigger&&!out.is_empty()&&!self.triggered.replace(true){if self.tamper{let last=out.len()-1;out[last]^=1;}else{return Err(Error::Io("synthetic readback primary".into()));}}
  Ok(())
 }
 fn append(&mut self,b:&[u8])->Result<u64>{self.inner.append(b)}
 fn write_at(&mut self,at:u64,b:&[u8])->Result<()>{if self.triggered.get()&&self.cleanup==Some(0){return Err(Error::Io("synthetic cleanup zero".into()));}self.inner.write_at(at,b)}
 fn truncate(&mut self,n:u64)->Result<()>{if self.triggered.get()&&self.cleanup==Some(2){return Err(Error::Io("synthetic cleanup truncate".into()));}self.inner.truncate(n)?;self.truncated.set(true);Ok(())}
 fn sync(&self)->Result<()>{if self.triggered.get()&&(self.cleanup==Some(1)||(self.cleanup==Some(3)&&self.truncated.get())){return Err(Error::Io("synthetic cleanup sync".into()));}self.inner.sync()}
}

#[derive(Debug)]
struct SwitchingDestination {
 inner:Guarded<SharedStore>,source:SharedStore,successor:std::cell::RefCell<Option<Vec<u8>>>,point:u8,
 dependencies:Cell<bool>,anchors:Cell<usize>,switched:Rc<Cell<bool>>,
}
impl Clone for SwitchingDestination{fn clone(&self)->Self{panic!("copy must not clone destination")}}
impl SwitchingDestination{fn switch(&self){if let Some(bytes)=self.successor.borrow_mut().take(){*self.source.0.borrow_mut()=StorageBackend::memory(bytes);self.switched.set(true);}}}
impl Storage for SwitchingDestination{
 fn len(&self)->Result<u64>{self.inner.len()}
 fn read_at(&self,at:u64,len:usize)->Result<Vec<u8>>{self.inner.read_at(at,len)}
 fn read_at_into(&self,at:u64,out:&mut[u8])->Result<()>{self.inner.read_at_into(at,out)}
 fn append(&mut self,b:&[u8])->Result<u64>{let at=self.inner.append(b)?;if self.point==0{self.switch();}Ok(at)}
 fn write_at(&mut self,at:u64,b:&[u8])->Result<()>{self.inner.write_at(at,b)?;if at==8192||at==FAILURE_REGION+8192{self.dependencies.set(true);}
 if (at==0||at==FAILURE_REGION)&&b.starts_with(b"RV4SHR01"){self.anchors.set(self.anchors.get()+1);}Ok(())}
 fn truncate(&mut self,n:u64)->Result<()>{self.inner.truncate(n)}
 fn sync(&self)->Result<()>{self.inner.sync()?;if (self.point==1&&self.dependencies.get())||(self.point==2&&self.anchors.get()==2){self.switch();}Ok(())}
}
