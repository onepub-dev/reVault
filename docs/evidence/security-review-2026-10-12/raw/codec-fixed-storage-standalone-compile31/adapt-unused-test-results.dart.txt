#! /usr/bin/env dcli
import 'dart:convert';
import 'dart:io';
import 'package:dcli/dcli.dart';
void main(List<String> args){
 const repo='/home/bsutton/git/.codex.workspaces/revault-issue-310-zip-read-performance';
 const snapshot='/tmp/revault-codec-fixed-storage-compile30/crate';
 final log=File('$repo/docs/evidence/security-review-2026-10-12/raw/codec-fixed-storage-standalone-compile30/tests.json');
 final edits=<String,Map<int,Map<String,dynamic>>>{};
 for(final line in log.readAsLinesSync()){if(!line.startsWith('{'))continue;final row=jsonDecode(line);if(row['reason']!='compiler-message')continue;final m=row['message'];if(m['level']!='error'||m['code']?['code']!='unused_must_use')continue;final spans=(m['spans'] as List).where((x)=>x['is_primary']==true).toList();if(spans.length!=1)throw StateError('Ambiguous primary');final p=Map<String,dynamic>.from(spans.single);final path=p['file_name'] as String;if(!path.startsWith('src/')&&!path.startsWith('tests/'))throw StateError('Unexpected path');edits.putIfAbsent(path,()=>{})[p['byte_end']]=p;}
 for(final e in edits.entries){final file=File('$repo/rust/vendor/zstd-complete/${e.key}');final current=file.readAsBytesSync(),bytes=File('$snapshot/${e.key}').readAsBytesSync();if(!args.contains('--verify')&&base64Encode(current)!=base64Encode(bytes))throw StateError('Source changed since diagnostic: ${e.key}');var updated=bytes.toList();final spans=e.value.values.toList()..sort((a,b)=>(b['byte_end'] as int).compareTo(a['byte_end'] as int));var previousStart=bytes.length;print('${e.key}: ${spans.length}');for(final p in spans){final a=p['byte_start'] as int,b=p['byte_end'] as int;if(a<0||b<a||b>previousStart)throw StateError('Overlapping or invalid diagnostic spans');previousStart=a;final expression=utf8.decode(bytes.sublist(a,b));if(expression.trimRight().endsWith(';'))throw StateError('Span includes semicolon');print('  ${p['line_start']}: ${expression.replaceAll(RegExp(r"\s+"),' ')}');updated.insertAll(b,utf8.encode('.expect("test storage admission")'));}if(args.contains('--verify')&&base64Encode(current)!=base64Encode(updated))throw StateError('Unexpected edited bytes: ${e.key}');if(args.contains('--apply'))file.writeAsBytesSync(updated);}
 print(green(args.contains('--verify')?'Verified disjoint exact diagnostic edits; no source mutation.':args.contains('--apply')?'Applied exact diagnostic test admission assertions.':'Review-only compiler spans; no source edits.'));
}
