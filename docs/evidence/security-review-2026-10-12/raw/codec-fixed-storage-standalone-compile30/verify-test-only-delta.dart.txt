#! /usr/bin/env dcli
import 'dart:io';
import 'package:dcli/dcli.dart';
void main(){
 const before='/tmp/revault-codec-fixed-storage-compile28/crate/src';
 const after='/home/bsutton/git/.codex.workspaces/revault-issue-310-zip-read-performance/rust/vendor/zstd-complete/src';
 var changed=0;
 Set<String> inventory(String root)=>Directory(root).listSync(recursive:true).whereType<File>().where((f)=>f.path.endsWith('.rs')).map((f)=>f.path.substring(root.length+1)).toSet();
 final oldInventory=inventory(before),newInventory=inventory(after);
 if(oldInventory.difference(newInventory).isNotEmpty||newInventory.difference(oldInventory).isNotEmpty)throw StateError('Rust file inventory changed');
 print('Rust file inventory unchanged: ${oldInventory.length} files.');
 for(final f in Directory(before).listSync(recursive:true).whereType<File>().where((f)=>f.path.endsWith('.rs'))){final rel=f.path.substring(before.length+1);final old=f.readAsStringSync(),now=File('$after/$rel').readAsStringSync();if(old==now)continue;changed++;
 if(rel.endsWith('/tests.rs')||rel.endsWith('_tests.rs')){print('test file: $rel');continue;}
 final marker=rel=='encoding/levels/c_port/post_split.rs'?'#[cfg(test)]\nfn sequence_prefixes':rel=='fse/fse_encoder/table.rs'?'#[cfg(test)]\nmod access_tests {':'#[cfg(test)]\nmod tests {';
 final a=old.indexOf(marker),b=now.indexOf(marker);if(a<0||b<0||old.substring(0,a)!=now.substring(0,b))throw StateError('Non-test delta requires review: $rel');
 print('unchanged production prefix, test suffix: $rel');
 }
 print(green('$changed changed Rust files classified by reviewed test paths/boundaries; prefixes equal passing compile28 snapshot. Suffix item reachability requires accompanying source review.'));
}
