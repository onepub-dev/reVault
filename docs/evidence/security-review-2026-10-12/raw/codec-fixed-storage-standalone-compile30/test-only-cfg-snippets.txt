   510	fn encode_partition(
   511	    block: &[u8],
   512	    last_block: bool,

   533	        context.policy,

   536	            literals: prepared.literals,
   537	            sequences: &resolved,
   538	        },
   539	        decompression_offsets,
   540	        context.previous_huff_table,
   541	        Some(&mut huffman_scratch),
   542	        Some(&mut fse_scratch),
   543	        context.fse_tables,
   544	        context.offset_history,
   545	        &mut bytes,
   546	    ).expect("test storage admission");

   550	            repeat_offsets: decompression_offsets,

   557	        },
