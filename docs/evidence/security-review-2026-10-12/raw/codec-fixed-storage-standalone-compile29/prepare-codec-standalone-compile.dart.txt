#! /usr/bin/env dcli
import 'dart:convert';
import 'dart:io';
import 'package:dcli/dcli.dart';
void main(){
 const source='/home/bsutton/git/.codex.workspaces/revault-issue-310-zip-read-performance/rust/vendor/zstd-complete';
 const dest='/tmp/revault-codec-fixed-storage-compile29';
 if(Directory(dest).existsSync())throw StateError('Refuse existing snapshot');
 Directory('$dest/crate').createSync(recursive:true);final rows=<Map<String,Object>>[];var total=0;
 final files=<File>[];for(final name in ['Cargo.toml','Cargo.lock','Readme.md'])files.add(File('$source/$name'));
 for(final name in ['src','tests','dict_tests','decodecorpus_files']){for(final e in Directory('$source/$name').listSync(recursive:true,followLinks:false)){if(e is Link)throw StateError('Unexpected source link');if(e is File)files.add(e);}}
 files.sort((a,b)=>a.path.compareTo(b.path));
 for(final file in files){final rel=file.path.substring(source.length+1);final length=file.lengthSync();total+=length;if(total>256*1024*1024)throw StateError('Source snapshot exceeds bound');final target=File('$dest/crate/$rel');target.parent.createSync(recursive:true);file.copySync(target.path);final oldHash=Process.runSync('/usr/bin/sha256sum',[file.path]),newHash=Process.runSync('/usr/bin/sha256sum',[target.path]);if(oldHash.exitCode!=0||newHash.exitCode!=0)throw StateError('Hash failure');final a=(oldHash.stdout as String).split(' ').first,b=(newHash.stdout as String).split(' ').first;if(a!=b)throw StateError('Copy mismatch');rows.add({'path':rel,'bytes':length,'sha256':a});}
 final manifest=File('$dest/crate/Cargo.toml');final original=manifest.readAsStringSync();if(original.contains('[workspace]'))throw StateError('Unexpected workspace table');File('$dest/Cargo.toml.original').writeAsStringSync(original);manifest.writeAsStringSync('$original\n[workspace]\n');
 File('$dest/source-snapshot.json').writeAsStringSync(const JsonEncoder.withIndent('  ').convert({'source':source,'destination':'$dest/crate','total_bytes':total,'files':rows,'scaffolding':'Only Cargo.toml appends an empty standalone [workspace]; existing package resolver 3/version/dependencies unchanged. Compiler-only; no runtime inputs.'})+'\n');
 print(green('Standalone compiler snapshot prepared: ${files.length} files, $total bytes.'));
}
