#! /usr/bin/env dcli
import 'dart:io';
import 'package:dcli/dcli.dart';
void main(List<String> args) {
 if(args.length!=1) throw ArgumentError('pass release test executable');
 const base='/tmp/revault-selective-read-20261010';
 const rust='/home/bsutton/git/.codex.workspaces/revault-issue-310-zip-read-performance/rust';
 Directory('$base/bin').createSync(recursive:true);
 File('/tmp/revault-tree-leaf-validation-20261010/bin/before').copySync('$base/bin/before');
 File(args.single).copySync('$base/bin/revault_lockbox_api_tests');
 File('/tmp/revault-tree-read-combined-corrected-20261010/bin/archive_evaluation').copySync('$base/bin/archive_evaluation');
 for(final name in ['small512-mixed-compressed','raw8m-random','compressed8m-pattern','raw64m-random']) {
  final original='/tmp/revault-tree-read-20261010-final/cases/$name';
  final fixture='$base/fixtures/$name';
  Directory('$fixture/other').createSync(recursive:true);
  for(final entry in {
   'archive.zip':'$original/archive.zip', 'inventory.json':'$original/inventory.json',
   'case.json':'$original/case.json', 'source':'$original/source',
   'candidate.lbox':'$original/other/tree.lbox', 'candidate.public':'$original/other/tree.public',
   'other/case.json':'$original/other/case.json', 'other/source':'$original/other/source',
   'other/tree.lbox':'$original/other/tree.lbox', 'other/tree.public':'$original/other/tree.public',
   'other/candidate.lbox':'$original/other/tree.lbox', 'other/candidate.public':'$original/other/tree.public',
  }.entries) { Link('$fixture/${entry.key}').createSync(entry.value); }
 }
 for(final variant in ['before','after']) {
  final binary=variant=='before'?'$base/bin/before':'$base/bin/revault_lockbox_api_tests';
  final environment=Map<String,String>.from(Platform.environment)..remove('REVAULT_CANDIDATE_FROZEN_DENSE')..remove('REVAULT_TREE_OPEN_DIAGNOSTIC');
  environment['REVAULT_CANDIDATE_FROZEN_TEST_BINARY']=binary;
  environment['REVAULT_CANDIDATE_FROZEN_TREE']='1';
  final result=Process.runSync('rustc',['+1.88.0','--edition=2021','-O','$rust/revault_lockbox_api/benches/evaluation/candidate_driver.rs','-o','$base/bin/adapter_$variant'],environment:environment);
  File('$base/adapter-$variant-build.log').writeAsStringSync('${result.stdout}${result.stderr}');
  if(result.exitCode!=0) throw StateError('adapter $variant failed');
 }
 var script=File('/tmp/revault-tree-read-combined-corrected-20261010/run-fixed-batch.dart').readAsStringSync();
 script=script.replaceAll('/tmp/revault-tree-read-combined-corrected-20261010',base).replaceAll('adapter_packed','adapter_before').replaceAll('adapter_tree','adapter_after');
 script=script.replaceFirst('/tmp/revault-tree-read-20261010-final/cases','$base/fixtures');
 script=script.replaceFirst("caseRoot, packed, caseRoot, tree, '\$caseRoot/other'", "caseRoot, packed, '\$caseRoot/other', tree, '\$caseRoot/other'");
 script=script.replaceFirst("final before = hashes();", "if (Platform.environment.containsKey('REVAULT_TREE_OPEN_DIAGNOSTIC')) throw StateError('normal reads only');\n  final before = hashes();");
 script=script.replaceFirst("'core_test_binary': core,", "'core_test_binary': core,\n  'baseline_test_binary': '\$root/bin/before',\n  'selective_reader': '\$rust/revault_lockbox_api/src/file_format/candidate_files/tree_image/reader.rs',\n  'page_cache': '\$rust/revault_lockbox_api/src/file_format/authenticated_index/cache.rs',");
 File('$base/run-fixed-batch.dart').writeAsStringSync(script);
 final chmod=Process.runSync('chmod',['+x','$base/run-fixed-batch.dart']);
 if(chmod.exitCode!=0) throw StateError('chmod failed');
 print('Prepared fixed six-case ZIP/eager/selective comparison. No timing run yet.');
}
