void save(String p,Object v)=>File(p).writeAsStringSync('${const JsonEncoder.withIndent('  ').convert(v)}\n',flush:true);
Future<Map<String,Object?>> runPhase(String name,List<String> args)async{
 final out=File('$evidence/$name.stdout').openWrite(),err=File('$evidence/$name.stderr').openWrite();final sw=Stopwatch()..start();
 final env={'PATH':'$toolbin:/home/bsutton/.cargo/bin:/usr/bin:/bin','TMPDIR':'/home/bsutton/.cache/rv4-security-20261011'};
 final p=await Process.start('/usr/bin/timeout',['--kill-after=30s','${cap}s',...args],workingDirectory:rust,environment:env,includeParentEnvironment:true);
 File('$evidence/$name.pid').writeAsStringSync('${p.pid}\n',flush:true);print(green('$name started pid=${p.pid} cap=${cap}s'));
 final a=p.stdout.listen(out.add).asFuture<void>(),b=p.stderr.listen(err.add).asFuture<void>();final code=await p.exitCode;await Future.wait([a,b]);await out.close();await err.close();sw.stop();
 final m={'argv':['/usr/bin/timeout','--kill-after=30s','${cap}s',...args],'cwd':rust,'pid':p.pid,'exit_code':code,'elapsed_seconds':sw.elapsedMilliseconds/1000,'timeout':code==124||code==137};save('$evidence/$name.status.json',m);print(green('$name completed exit=$code elapsed=${sw.elapsedMilliseconds/1000}s'));return m;
}
String? passedCount(String p){final s=File(p).readAsStringSync();final m=RegExp(r'test result: ok\. (\d+) passed').firstMatch(s);return m?.group(1);}
void post(){final r=Process.runSync('/usr/bin/git',['rev-parse','HEAD'],workingDirectory:worktree);final head=(r.stdout as String).trim();File('$evidence/head-post.txt').writeAsStringSync('$head\n');final files=Process.runSync('/usr/bin/git',['ls-files','*.rs','Cargo.toml','Cargo.lock'],workingDirectory:rust);if(files.exitCode!=0)throw StateError('git ls-files failed');final paths=(files.stdout as String).split('\n').where((x)=>x.isNotEmpty).toList()..sort();final lines=<String>[];for(final p in paths){final h=Process.runSync('/usr/bin/sha256sum',['--','$rust/$p']);if(h.exitCode!=0)throw StateError('hash failed $p');lines.add('${(h.stdout as String).split(RegExp(r'\s+')).first}  $p');}File('$evidence/source-manifest-post.sha256').writeAsStringSync('${lines.join('\n')}\n');save('$evidence/postterminal.json',{'head':head,'source_manifest_entries':lines.length,'source_manifest_unchanged':File('$evidence/source-manifest-pre.sha256').readAsStringSync()=='${lines.join('\n')}\n'});}
