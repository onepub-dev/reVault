use std::{error::Error, fs, io};
use revault_page_api::{allocation_chunk_bytes, weakened_allocation_allowed, Error as ApiError, SecureVec};
const LIMIT:u64=8*1024*1024;
fn status(key:&str)->Result<u64,Box<dyn Error>>{let s=fs::read_to_string("/proc/self/status")?;let l=s.lines().find(|x|x.starts_with(&format!("{key}:"))).ok_or_else(||io::Error::other(format!("missing {key}")))?;Ok(l.split_whitespace().nth(1).ok_or_else(||io::Error::other("missing status value"))?.parse()?)}
fn limits()->Result<(u64,u64),Box<dyn Error>>{let s=fs::read_to_string("/proc/self/limits")?;let f=s.lines().find(|x|x.starts_with("Max locked memory")).ok_or_else(||io::Error::other("missing memlock limit"))?.split_whitespace().collect::<Vec<_>>();if f.len()!=6||f[5]!="bytes"{return Err(io::Error::other("unexpected memlock limit format").into())}Ok((f[3].parse()?,f[4].parse()?))}
fn row(variant:&str,pass:usize,index:usize,size:usize,stage:&str)->Result<u64,Box<dyn Error>>{let locked=status("VmLck")?.checked_mul(1024).ok_or_else(||io::Error::other("VmLck overflow"))?;let rss=status("VmRSS")?;let(soft,hard)=limits()?;if soft!=LIMIT||hard!=LIMIT{return Err(io::Error::other(format!("memlock changed {soft}/{hard}")).into())}if locked>LIMIT{return Err(io::Error::other(format!("VmLck exceeds limit: {locked}")).into())}if stage=="baseline"&&locked!=0{return Err(io::Error::other(format!("baseline VmLck {locked} != 0")).into())}println!("{{\"variant\":\"{variant}\",\"pass\":{pass},\"index\":{index},\"stage\":\"{stage}\",\"requested_bytes\":{size},\"capacity_bytes_source_inferred\":{size},\"VmLck_bytes\":{locked},\"VmRSS_kib\":{rss},\"memlock_soft_bytes\":{soft},\"memlock_hard_bytes\":{hard},\"weak_policy\":false}}");Ok(locked)}
fn run(variant:&str)->Result<(),Box<dyn Error>>{if weakened_allocation_allowed(){return Err(io::Error::other("weak allocation policy must remain false").into())}if allocation_chunk_bytes()!=65536{return Err(io::Error::other(format!("allocation_chunk_bytes {} != 65536",allocation_chunk_bytes())).into())}if std::env::var("PROFILE_HOST_PAGE_SIZE").ok().as_deref()!=Some("4096"){return Err(io::Error::other("host page size not verified as 4096").into())}row(variant,0,0,0,"baseline")?;let mut count=0usize;for pass in 1..=2{for index in 0..64{let size=8192+index*4096;row(variant,pass,index,size,"before")?;let byte=(((index+(pass-1)*64)%255)+1)as u8;let input=vec![byte;size];let value=match SecureVec::try_from_slice(&input){Ok(v)=>v,Err(e)if variant=="old"&&e==ApiError::WeakAllocationDisabled=>{let locked=row(variant,pass,index,size,"expected_old_refusal")?;println!("{{\"variant\":\"old\",\"outcome\":\"expected_refusal_not_ignored\",\"pass\":{pass},\"index\":{index},\"requested_bytes\":{size},\"VmLck_bytes\":{locked},\"error\":\"WeakAllocationDisabled\"}}");return Ok(())},Err(e)=>return Err(e.into())};if value.len()!=size{return Err(io::Error::other("SecureVec length mismatch").into())}let matched=value.with_bytes(|bytes|bytes==input.as_slice())?;if !matched{return Err(io::Error::other("exact scoped byte comparison failed").into())}row(variant,pass,index,size,"live")?;drop(value);row(variant,pass,index,size,"after_drop")?;count+=1}}if variant=="old"{return Err(io::Error::other("old variant did not refuse under pressure").into())}if count!=128{return Err(io::Error::other(format!("new completed {count}, expected 128")).into())}println!("{{\"variant\":\"new\",\"outcome\":\"completed\",\"allocations\":{count},\"passes\":2}}");Ok(())}
fn main(){let variant=std::env::args().nth(1).unwrap_or_default();if variant!="old"&&variant!="new"{eprintln!("variant must be old or new");std::process::exit(2)}if let Err(e)=run(&variant){eprintln!("profile failure: {e}");std::process::exit(1)}}
