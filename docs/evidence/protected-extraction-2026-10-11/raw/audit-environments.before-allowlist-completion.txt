#! /usr/bin/env dcli
import 'dart:convert';import 'dart:io';import 'package:dcli/dcli.dart';
const root='/home/bsutton/.cache/revault-protected-20261011-evidence';
void main(){final keys=<String>{};var maps=0;void walk(dynamic x){if(x is Map){for(final e in x.entries){if(e.key=='environment'&&e.value is Map){maps++;keys.addAll((e.value as Map).keys.cast<String>());}walk(e.value);}}else if(x is List){for(final v in x)walk(v);}}
 for(final f in Directory(root).listSync(recursive:true,followLinks:false).whereType<File>()){final rel=f.path.substring(root.length+1);if(rel.split('/').any((s)=>['tmp','.dart_tool','provenance-reconstruction'].contains(s)))continue;if(!rel.endsWith('.json'))continue;walk(jsonDecode(f.readAsStringSync()));}
 final allowed={'REVAULT_EXTRACT_DURABLE','REVAULT_EXTRACT_TID_PARITY','REVAULT_EXTRACT_PID_NAMESPACE','REVAULT_EXTRACT_BLOCK_DEVICES','REVAULT_EXTRACT_WORKERS','REVAULT_ARCHIVE_WORKERS','REVAULT_TREE_EXTRACT_WORKERS'};final unexpected=keys.difference(allowed);print('Environment keys: ${keys.toList()..sort()}');if(unexpected.isNotEmpty)throw StateError('Unexpected key names: $unexpected');File('$root/environment-allowlist-audit.json').writeAsStringSync(jsonEncode({'passed':true,'environment_maps_checked':maps,'keys':keys.toList()..sort(),'scope':'explicit environment maps in retained JSON; no inherited environment dump'}),flush:true);print(green('Curated environment audit passed.'));}
