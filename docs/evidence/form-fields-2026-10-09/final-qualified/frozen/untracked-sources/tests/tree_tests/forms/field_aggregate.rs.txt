fn memory()->serde_json::Value {let s=std::fs::read_to_string("/proc/self/status").unwrap();let n=|name:&str|s.lines().find_map(|l|l.strip_prefix(name)).unwrap().split_whitespace().next().unwrap().parse::<u64>().unwrap();serde_json::json!({"vm_lck_kib":n("VmLck:"),"vm_rss_kib":n("VmRSS:"),"vm_hwm_kib":n("VmHWM:")})}
#[derive(Debug)]
struct NoClone<S>(Guarded<S>);
impl<S> Clone for NoClone<S>{fn clone(&self)->Self{panic!("field upgrade cloned underlying storage")}}
impl<S:Storage> Storage for NoClone<S>{
 fn len(&self)->Result<u64>{self.0.len()}
 fn read_at(&self,at:u64,len:usize)->Result<Vec<u8>>{self.0.read_at(at,len)}
 fn read_at_into(&self,at:u64,out:&mut[u8])->Result<()>{self.0.read_at_into(at,out)}
 fn append(&mut self,b:&[u8])->Result<u64>{self.0.append(b)}
 fn write_at(&mut self,at:u64,b:&[u8])->Result<()>{self.0.write_at(at,b)}
 fn truncate(&mut self,len:u64)->Result<()>{self.0.truncate(len)}
 fn sync(&self)->Result<()>{self.0.sync()}
}
#[test]
#[ignore="fixed fresh-process cross-record field upgrade aggregate probe"]
fn typed_form_field_upgrade_aggregate_probe(){

 {let def=FormDefinition{type_id:type_id.clone(),alias:"upgrade".into(),revision:1,name:"Aggregate upgrade".into(),description:"Nine independent values".into(),fields:vec![FormFieldDefinition{id:"value".into(),label:"Current label".into(),kind:FormFieldKind::Notes,required:false}]};

 let (old,old_def)={let opened=TreeImage::open(crate::file_format::allocation_map::compaction::View(&writer),archive(),mode,&authority,key(mode)).unwrap();(opened.image.catalogue.forms.records.iter().flat_map(|r|r.fields.iter().flat_map(|f|f.label.extents.iter().chain(&f.value.extents))).copied().collect::<Vec<_>>(),{let d=&opened.image.catalogue.forms.definitions[0];std::iter::once(&d.name).chain(std::iter::once(&d.description)).chain(d.fields.iter().map(|f|&f.label)).cloned().collect::<Vec<_>>()})};
 let value=FormValue::secret(SecretString::try_from_slice(&vec![b'z';size]).unwrap());let after_caller=memory();

 let ranges=writer.0.spans.borrow().clone();drop(writer);drop(value);
 let mut opened=TreeImage::open(NoClone(Guarded::new(StorageBackend::file(&path).unwrap(),ranges)),archive(),mode,&authority,key(mode)).unwrap();opened.image.verify_all().unwrap();
 {let d=opened.image.catalogue.forms.definitions.iter().find(|d|d.revision==1).unwrap();assert_eq!(std::iter::once(&d.name).chain(std::iter::once(&d.description)).chain(d.fields.iter().map(|f|&f.label)).cloned().collect::<Vec<_>>(),old_def);}
 for revision in [1,2] {let d=opened.get_form_definition(&type_id,revision).unwrap().unwrap();assert_eq!(d.name,"Aggregate upgrade");assert_eq!(d.description,"Nine independent values");assert_eq!(d.alias,"upgrade");assert_eq!(d.fields[0].kind,if revision==1{FormFieldKind::Notes}else{FormFieldKind::Secret});assert_eq!(d.fields[0].label,"Current label");}
 for n in 0..9 {let p=LockboxPath::new(format!("/records/{n}")).unwrap();let r=opened.image.catalogue.forms.records.iter().find(|r|r.path==p).unwrap();assert_eq!(r.revision,2);assert_eq!(r.alias,"upgrade");

 let texts=opened.image.catalogue.forms.texts();let logical_bytes:usize=texts.iter().map(|t|t.length).sum();let stored_payload_bytes:u64=texts.iter().flat_map(|t|&t.extents).map(|e|e.len).sum();assert!(logical_bytes>8*size&&stored_payload_bytes>8*size as u64);let after_read=memory();

 println!("FORM_FIELD_AGGREGATE {}",serde_json::json!({"mode":bits,"logical_bytes":logical_bytes,"stored_payload_bytes":stored_payload_bytes,"archive_before":archive_before,"archive_bytes":archive_after,"before_setup":before_setup,"after_setup":after_setup,"after_caller":after_caller,"after_upgrade":after_upgrade,"after_read":after_read,"after_salvage":after_salvage}));
}
